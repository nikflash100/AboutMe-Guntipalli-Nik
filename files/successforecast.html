#Tune the top 3 baselines with GridSearchCV (5-fold inside)
top_3 = baseline_df.head(3).index.tolist()
print("Tuning:", top_3)

param_grids = {
    'Logistic Regression': {'C': [0.01, 0.1, 1, 10], 'penalty': ['l2']},
    'KNN':                 {'n_neighbors': [3, 5, 7, 11, 15], 'weights': ['uniform', 'distance']},
    'Naive Bayes':         {'var_smoothing': [1e-9, 1e-8, 1e-7]},
    'Decision Tree':       {'max_depth': [3, 5, 7, 10, None], 'min_samples_split': [2, 5, 10]},
    'Random Forest':       {'n_estimators': [100, 200], 'max_depth': [5, 10, None], 'min_samples_split': [2, 5]},
    'SVM':                 {'C': [0.1, 1, 10], 'kernel': ['rbf', 'linear']},
    'Gradient Boosting':   {'n_estimators': [100, 200], 'learning_rate': [0.05, 0.1], 'max_depth': [3, 5]},
    'Neural Net':          {'hidden_layer_sizes': [(50,), (100,), (50, 50)], 'alpha': [0.0001, 0.001]},
}

tuned = {}
for name in top_3:
    model, needs_scale = models[name]
    X_use = X_train_scaled if needs_scale else X_train
    grid = GridSearchCV(model, param_grids[name], cv=5, scoring='accuracy', n_jobs=-1)
    grid.fit(X_use, y_train)
    tuned[name] = (grid.best_estimator_, needs_scale)
    print(f"\n{name}")
    print(f"  best CV acc: {grid.best_score_:.4f}")
    print(f"  best params: {grid.best_params_}")

Voting Classifier¶

In [ ]:
#Soft voting averages predicted probabilities from each model
#Use scaled features uniformly so all base estimators get the same input
voting_estimators = [(name.replace(' ', '_'), model) for name, (model, _) in tuned.items()]
voting = VotingClassifier(estimators=voting_estimators, voting='soft')
voting.fit(X_train_scaled, y_train)

v_scores = cross_val_score(voting, X_train_scaled, y_train, cv=5, scoring='accuracy')
print(f"Voting Classifier CV acc: {v_scores.mean():.4f} (+/- {v_scores.std():.4f})")

#Add voting to the tuned dict for final evaluation

Final evaluation¶

In [ ]:
#Evaluate every tuned model + voting on the TEST set
def evaluate(model, X_t, y_t):
    y_pred = model.predict(X_t)
    y_proba = model.predict_proba(X_t)[:, 1]
    return {
        'accuracy':  accuracy_score(y_t, y_pred),
        'precision': precision_score(y_t, y_pred),
        'recall':    recall_score(y_t, y_pred),
        'f1':        f1_score(y_t, y_pred),
        'roc_auc':   roc_auc_score(y_t, y_proba),
    }

test_results = {}
for name, (model, needs_scale) in tuned.items():
    X_use = X_test_scaled if needs_scale else X_test
    test_results[name] = evaluate(model, X_use, y_test)

test_df = pd.DataFrame(test_results).T.sort_values('accuracy', ascending=False)
test_df.round(4)
